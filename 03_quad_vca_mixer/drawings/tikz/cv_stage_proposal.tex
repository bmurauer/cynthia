\begin{circuitikz}[font=\sffamily\small, show background rectangle,

  \tikzset{pin/.style={font=\sffamily\scriptsize, text=gray}}

  % ---- jack and level pot ----

  \draw (top) to[pR, n=pot, l_={\shortstack{RV1 LEVEL\\100\,k$\Omega$ log}}] (2,-1.5) node[ground]{};

  % ---- inverting stage ----
  \node[op amp] (u) at (8,-0.49) {\scriptsize U3A};

  \node[pin, below left] at (u.+) {3};
  \node[pin, above right] at (u.out) {1};
  \draw (pot.wiper) -- (pot.wiper -| 3.5,0) to[R, l={220\,k$\Omega$}, -*] (u.- -| 6,0) coordinate (inv) -- (u.-);
  \draw (u.+) -- ++(-0.4,0) -- ++(0,-0.6) node[ground]{};
  \draw (inv) -- (6,2.5) to[R, l={330\,k$\Omega$}] (8,2.5)
        to[vR, l={RV5 100\,k$\Omega$}, a={CV scale}] (10,2.5) -- (10,0 |- u.out);

  \draw (e) -- (13,1.6) node[above]{TP3};

  \node[font=\sffamily\scriptsize, align=left, anchor=north west] at (3.0,-2.4)
     {$V_\mathrm{out} = -V_\mathrm{wiper}\cdot(330\,\mathrm{k} + \mathrm{RV5})/220\,\mathrm{k}$\\

  \node[font=\sffamily\scriptsize, align=left, anchor=north west] at (13.4,-1.4)
     {negative CV $\rightarrow$ U3A out positive\\$\rightarrow$ D3 clamps, channel off};

  \draw (0,-5) node[vcc]{+12\,V} to[R, l_={13\,k$\Omega$}] (0,-7) coordinate (d)

  \node[op amp, noinv input up] (b) at (3.5,-7.49) {\scriptsize U1C};
  \draw (d) to[short, *-] (d -| b.+) -- (b.+);
  \draw (b.-) -- ++(0,-1) -| (b.out);
  \draw (b.out) to[short, -o] ++(1.5,0) node[right]{VREF $\approx$ 5\,V (to TN of all CV jacks)};
